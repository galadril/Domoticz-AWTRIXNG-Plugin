-- Advances the AWTRIX NG app rotation on a timer.
--
-- Replace 'AWTRIXNG' below with whatever you named the hardware in
-- Setup -> Hardware.
return {
    -- The script is active
    active = true,

    -- Trigger section: run the script every 1 minute
    on = {
        timer = {
            'every 1 minutes'
        }
    },

    -- Custom logging level for this script
    logging = {
        level = domoticz.LOG_INFO,
        marker = "AWTRIX_NEXT"
    },

    -- Event execution when the timer triggers the script
    execute = function(domoticz)
        local appNextDevice = domoticz.devices('AWTRIXNG - Next App')
        appNextDevice.switchOn()
    end
}
